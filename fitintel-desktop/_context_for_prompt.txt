import requests
from typing import Optional, Dict, Any, List


def _rs(resp):
    """raise_for_status с человекочитаемым detail от сервера."""
    try:
        resp.raise_for_status()
    except Exception:
        try:
            d = resp.json().get("detail")
            if d:
                raise RuntimeError(str(d))
        except RuntimeError:
            raise
        except Exception:
            pass
        raise





def _check(resp):
    """HTTP-ошибка -> человеческий текст из detail сервера."""
    if resp.status_code >= 400:
        msg = "Ошибка сервера (код %s)" % resp.status_code
        try:
            d = resp.json()
            det = d.get("detail") if isinstance(d, dict) else None
            if det:
                msg = str(det)
        except Exception:
            pass
        raise RuntimeError(msg)

class ApiClient:
    def __init__(self, base_url: str = "http://localhost:8001/api/v1"):
        self.base_url = base_url.rstrip("/")
        self.token: Optional[str] = None
        self.session = requests.Session()
        self.session.headers.update({"Content-Type": "application/json"})
        try:
            from app_logging import log as _log
            def _resp_hook(resp, *args, **kwargs):
                if resp.status_code >= 400:
                    _log.warning("%s %s -> %s: %s", resp.request.method, resp.url,
                                 resp.status_code, resp.text[:200])
            self.session.hooks["response"].append(_resp_hook)
        except Exception:
            pass

        # Таймаут по умолчанию (10 сек) для ВСЕХ запросов сессии.
        _orig_request = self.session.request
        def _request_with_timeout(method, url, **kwargs):
            kwargs.setdefault("timeout", 10)
            return _orig_request(method, url, **kwargs)
        self.session.request = _request_with_timeout

    def set_token(self, token: str):
        self.token = token
        self.session.headers["Authorization"] = f"Bearer {token}"

    def clear_token(self):
        self.token = None
        self.session.headers.pop("Authorization", None)

    @staticmethod
    def _as_list(data) -> List[Dict[str, Any]]:
        """API может вернуть list или dict с вложенным списком."""
        if isinstance(data, list):
            return data
        if isinstance(data, dict):
            for k in ("items", "data", "results", "visits", "clients",
                      "subscriptions", "logs", "records", "entries", "users",
                      "devices", "drivers"):
                if isinstance(data.get(k), list):
                    return data[k]
        return []

    def _url(self, path: str) -> str:
        return f"{self.base_url}/{path.lstrip('/')}"

    def login(self, username: str, password: str) -> Dict[str, Any]:
        payload = {"login": username, "password": password}
        resp = self.session.post(self._url("/auth/login"), json=payload)
        _check(resp)
        data = resp.json()
        # сохраняем refresh_token сразу (E70 будет обновлять сессию молча)
        rt = data.get("refresh_token")
        if rt:
            self._refresh_token = rt
            _e70_save_tokens(getattr(self, "token", None), rt)
        return data

    def me(self) -> Dict[str, Any]:
        resp = self.session.get(self._url("/auth/me"))

import requests as _rq

def _e55_url(self, path):
    base = (getattr(self, "base_url", None) or getattr(self, "base", None)
            or getattr(self, "BASE_URL", None) or "http://localhost:8001")
    base = str(base).rstrip("/")
    if base.endswith("/api/v1"):
        return base + path
    return base + "/api/v1" + path

def _e55_sess(self):
    return getattr(self, "session", None) or getattr(self, "_session", None) or _rq

def _e55_get(self, path, params=None):
    r = _e55_sess(self).get(_e55_url(self, path), params=params or {}, timeout=60)
    _rs(r)
    try: return r.json()
    except Exception: return r.text

def _e55_post(self, path, payload=None, files=None, data=None):
    s = _e55_sess(self)
    if files is not None:
        r = s.post(_e55_url(self, path), files=files, data=data or {}, timeout=180)
    else:
        r = s.post(_e55_url(self, path), json=payload or {}, timeout=180)
    _rs(r)
    try: return r.json()
    except Exception: return r.text

def _e55_list(self, data):
    f = getattr(self, "_as_list", None)

            if isinstance(data.get(k), list): return data[k]
        return []
    return data if isinstance(data, list) else []

def _e55_try(self, kind, paths, payload=None):
    last = None
    for pth in paths:
        try:
            if kind == "get": return _e55_get(self, pth)
            if kind == "post": return _e55_post(self, pth, payload)
        except Exception as e:
            last = e
    raise last

def _e55_patch(self, path, payload=None):
    import requests
    url = _e55_url(self, path)
    if hasattr(self, "_session"):
        r = self._session.patch(url, json=payload or {}, timeout=30)
    else:
        r = requests.patch(url, json=payload or {}, timeout=30)
    _rs(r)
    try:
        return r.json()
    except Exception:
        return {}

# ============================ E70_EXT: прозрачное обновление сессии ============================
"""
Схема: access-токен короткий (30 мин). Любой запрос с 401 -> один молчаливый
refresh (refresh_token из client_settings.json, ротация на сервере) -> повтор
исходного запроса. Пользователь не видит «Токен истёк». Если refresh мёртв —
fallback на логин/пароль (если сохранены), иначе 401 доходит до UI (разлогин).
"""
import json as _e70_json
from pathlib import Path as _e70_Path


def _e70_settings_path():
    try:
        from paths import SETTINGS as _p
        return _p
    except Exception:
        return _e70_Path(__file__).resolve().parents[1] / "client_settings.json"


def _e70_load_settings() -> dict:
    try:
        p = _e70_settings_path()
        if p.exists():
            return _e70_json.loads(p.read_text(encoding="utf-8"))
    except Exception:
        pass
    return {}


def _e70_save_tokens(token, refresh_token):
    try:
        p = _e70_settings_path()
        cfg = _e70_load_settings()
        if token:
            cfg["token"] = token
            cfg["access_token"] = token
        if refresh_token:
            cfg["refresh_token"] = refresh_token
        p.write_text(_e70_json.dumps(cfg, ensure_ascii=False, indent=2),
                     encoding="utf-8")
    except Exception:
        pass


def _e70_refresh(self) -> bool:
    """Обмен refresh_token на новую пару. Молча, с сохранением в настройки."""
    rt = getattr(self, "_refresh_token", None) or _e70_load_settings().get("refresh_token")
    if not rt:
        return False
    try:
        r = self.session.post(self._url("/auth/refresh"),
                              json={"refresh_token": rt}, timeout=15)
        if r.status_code != 200:
            return False
        data = r.json()
        token = data.get("access_token")
        new_rt = data.get("refresh_token") or rt
        if not token:
            return False
        self._refresh_token = new_rt
        self.set_token(token)
        _e70_save_tokens(token, new_rt)
        try:
            from app_logging import log as _l
            _l.info("E70: сессия обновлена через refresh_token")

def _e70_relogin(self) -> bool:
    """Молчаливая повторная авторизация: сначала refresh, потом логин/пароль."""
    if getattr(self, "_e70_refreshing", False):
        return False
    self._e70_refreshing = True
    try:
        if _e70_refresh(self):
            return True

        cfg = _e70_load_settings()
        login = cfg.get("saved_login") or cfg.get("login") or cfg.get("username")
        password = cfg.get("saved_password") or cfg.get("password")
        if not login or not password:
            return False
        old_auth = self.session.headers.pop("Authorization", None)
        try:
            r = self.session.post(self._url("/auth/login"),
                                  json={"login": login, "password": password})
            if r.status_code != 200:
                return False
            data = r.json()
            token = data.get("access_token")
            rt = data.get("refresh_token")
        finally:
            if old_auth and "Authorization" not in self.session.headers:
                self.session.headers["Authorization"] = old_auth
        if not token:
            return False
        if rt:
            self._refresh_token = rt
        self.set_token(token)
        _e70_save_tokens(token, rt)
        try:
            from app_logging import log as _l
            _l.info("E70: сессия обновлена через логин/пароль")
        except Exception:
            pass
        return True
    except Exception:
        return False
    finally:
        self._e70_refreshing = False

                kwargs = dict(kwargs, headers=h)
            resp = _inner(method, url, **kwargs)

    self.session.request = _e70_request

    def _load_roles(self):
        """Роли из RBAC (без системных client/device и superadmin)."""
        try:
            resp = self.api.session.get(self.api._url("/rbac/roles-matrix"))
            resp.raise_for_status()
            body = resp.json()
            rows = body.get("roles", body) if isinstance(body, dict) else body
            skip = {"client", "device", "superadmin"}
            labels = {"admin": "Администратор", "manager": "Менеджер",
                      "trainer": "Тренер", "reception": "Ресепшен",
                      "cashier": "Кассир", "accountant": "Бухгалтер",
                      "support": "Поддержка", "owner": "Владелец"}
            out = []
            for r in rows if isinstance(rows, list) else []:
                code = str(r.get("role_code") or r.get("code") or "").lower()
                if not code or code in skip:
                    continue
                name = labels.get(code, r.get("role_name") or code)
                out.append((name, code))
            if out:
                return out
        except Exception:
            pass
        return [("Менеджер", "manager"), ("Тренер", "trainer"),

            return

        uid = item.data(Qt.ItemDataRole.UserRole) if item else None
        login = item.text() if item else "?"
        if not uid:
            QMessageBox.warning(self, "Удаление", "ID пользователя не найден — нажмите «Обновить»")
            return
        if login.lower() == "superadmin":
            QMessageBox.warning(self, "Удаление", "superadmin нельзя удалить")
            return
        if QMessageBox.question(

                QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No,
                QMessageBox.StandardButton.No) != QMessageBox.StandardButton.Yes:
            return
        try:
            r = self.api.session.delete(self.api._url(f"/users/{uid}"))
            r.raise_for_status()
            self.refresh()

            QMessageBox.critical(self, "Ошибка удаления", str(e))

"""FitIntel Pro — Роли и права: редактируемая матрица (галочки + сохранение)."""
from PyQt6.QtWidgets import (

    QTableWidget, QTableWidgetItem, QHeaderView, QSplitter, QAbstractItemView

def _err_text(e) -> str:
    resp = getattr(e, "response", None)
    if resp is not None:
        try:
            d = resp.json().get("detail")
            if d:
                return str(d)
        except Exception:
            pass
        return "Ошибка сервера %s" % resp.status_code
    return str(e)


class _Get(QThread):
    done = pyqtSignal(object)
    fail = pyqtSignal(object)
    def __init__(self, api, path):
        super().__init__()
        self.api, self.path = api, path
    def run(self):
        try:
            r = self.api.session.get(self.api._url(self.path), timeout=20)
            r.raise_for_status()
            self.done.emit(r.json())
        except Exception as e:
            self.fail.emit(e)


class RolesTab(QWidget):
    """Слева — роли, справа — все права с галочками. Сохранение = diff на сервер."""

    def __init__(self, api: ApiClient):
        super().__init__()
        self.api = api
        self._roles = []          # [{id, code, name, perms:set(perm_id)}]
        self._perms = []          # [{id, code, name}]
        self._edits = {}          # role_id -> set(perm_id) (рабочая копия)
        self._loading = False
        self._build_ui()
        self.refresh()

    # ---------- UI ----------
    def _build_ui(self):
        layout = QVBoxLayout(self)
        layout.setContentsMargins(16, 16, 16, 16)
        layout.setSpacing(10)

        top = QHBoxLayout()
        self.lbl_hint = QLabel("Выберите роль слева → отметьте права галочками → «💾 Сохранить»")
        top.addWidget(self.lbl_hint, 1)
        b = QPushButton(" Обновить")
        b.clicked.connect(self.refresh)
        top.addWidget(b)
        self.b_save = QPushButton(" Сохранить изменения")
        self.b_save.setEnabled(False)
        self.b_save.clicked.connect(self._save)
        top.addWidget(self.b_save)
        layout.addLayout(top)

        split = QSplitter(Qt.Orientation.Horizontal)

        self.tbl_roles = QTableWidget(0, 2)
        self.tbl_roles.setHorizontalHeaderLabels(["Роль", "Код"])
        self.tbl_roles.horizontalHeader().setSectionResizeMode(QHeaderView.ResizeMode.Stretch)
        self.tbl_roles.setEditTriggers(QAbstractItemView.EditTrigger.NoEditTriggers)
        self.tbl_roles.setSelectionBehavior(QAbstractItemView.SelectionBehavior.SelectRows)
        self.tbl_roles.setAlternatingRowColors(True)
        self.tbl_roles.setStyleSheet(theme.table_style())
        self.tbl_roles.itemSelectionChanged.connect(self._on_role_selected)
        split.addWidget(self.tbl_roles)

        self.tbl_perms = QTableWidget(0, 2)
        self.tbl_perms.setHorizontalHeaderLabels(["✓", "Право"])
        self.tbl_perms.setColumnWidth(0, 40)
        self.tbl_perms.horizontalHeader().setSectionResizeMode(1, QHeaderView.ResizeMode.Stretch)
        self.tbl_perms.setEditTriggers(QAbstractItemView.EditTrigger.NoEditTriggers)
        self.tbl_perms.setAlternatingRowColors(True)
        self.tbl_perms.setStyleSheet(theme.table_style())
        self.tbl_perms.itemChanged.connect(self._on_perm_toggled)
        split.addWidget(self.tbl_perms)
        split.setSizes([300, 520])
        layout.addWidget(split, 1)

        self.status = QLabel("")
        layout.addWidget(self.status)

    # ---------- загрузка ----------
    def refresh(self):
        self.status.setText("Загрузка ролей и прав…")
        self._loading = True
        w1 = _Get(self.api, "/roles/")
        w1.done.connect(self._roles_loaded)
        w1.fail.connect(self._load_fail)
        w1.start()
        self._w1 = w1

    def _roles_loaded(self, data):
        roles = data if isinstance(data, list) else []
        self._roles = []
        for r in roles:
            perms = set()
            for rp in (r.get("role_permissions") or []):
                p = rp.get("permission") if isinstance(rp, dict) else None
                if p and p.get("id"):
                    perms.add(str(p["id"]))
            self._roles.append({
                "id": str(r.get("id")),
                "code": r.get("code") or "",
                "name": r.get("name") or r.get("code") or "—",
                "perms": perms,
            })
        self._edits = {r["id"]: set(r["perms"]) for r in self._roles}
        w2 = _Get(self.api, "/permissions/")
        w2.done.connect(self._perms_loaded)
        w2.fail.connect(self._load_fail)
        w2.start()
        self._w2 = w2

    def _perms_loaded(self, data):
        perms = data if isinstance(data, list) else []
        self._perms = [{"id": str(p.get("id")), "code": p.get("code") or "",
                        "name": p.get("name") or p.get("code") or ""}
                       for p in perms if p.get("id")]
        self._render_roles()
        self._loading = False
        self.status.setText("Готово: %d ролей, %d прав. Права применяются сразу после сохранения."
                            % (len(self._roles), len(self._perms)))

    def _load_fail(self, e):
        self._loading = False
        self.status.setText("Ошибка загрузки: %s" % _err_text(e))

    # ---------- отрисовка ----------
    def _render_roles(self):
        self.tbl_roles.setRowCount(len(self._roles))
        for i, r in enumerate(self._roles):
            self.tbl_roles.setItem(i, 0, QTableWidgetItem(r["name"]))
            self.tbl_roles.setItem(i, 1, QTableWidgetItem(r["code"]))

    def _current_role(self):
        row = self.tbl_roles.currentRow()
        if 0 <= row < len(self._roles):
            return self._roles[row]
        return None

    def _on_role_selected(self):
        role = self._current_role()
        self._render_perms(role)

    def _render_perms(self, role):
        self._loading = True
        self.tbl_perms.setRowCount(len(self._perms))
        checked = self._edits.get(role["id"], set()) if role else set()
        for i, p in enumerate(self._perms):
            it = QTableWidgetItem()
            it.setData(Qt.ItemDataRole.UserRole, p["id"])
            it.setFlags(Qt.ItemFlag.ItemIsUserCheckable | Qt.ItemFlag.ItemIsEnabled)
            it.setCheckState(Qt.CheckState.Checked if p["id"] in checked
                             else Qt.CheckState.Unchecked)
            self.tbl_perms.setItem(i, 0, it)
            name = "%s — %s" % (p["code"], p["name"]) if p["name"] and p["name"] != p["code"] else p["code"]
            self.tbl_perms.setItem(i, 1, QTableWidgetItem(name))
        self._loading = False
        self.b_save.setEnabled(role is not None)

    def _on_perm_toggled(self, item):
        if self._loading:
            return
        role = self._current_role()
        if not role or item.column() != 0:
            return
        pid = item.data(Qt.ItemDataRole.UserRole)
        if item.checkState() == Qt.CheckState.Checked:
            self._edits[role["id"]].add(pid)
        else:
            self._edits[role["id"]].discard(pid)

    # ---------- сохранение ----------
    def _save(self):
        role = self._current_role()
        if not role:
            return
        if role["code"] == "admin":
            ret = QMessageBox.question(self, "Роль admin",
                "Вы редактируете системную роль admin. Продолжить?")
            if ret != QMessageBox.StandardButton.Yes:
                return
        orig, now = role["perms"], self._edits[role["id"]]
        to_add = [p for p in (now - orig)]
        to_del = [p for p in (orig - now)]
        if not to_add and not to_del:
            self.status.setText("Изменений нет.")
            return
        self.b_save.setEnabled(False)
        self.status.setText("Сохранение: +%d / −%d прав…" % (len(to_add), len(to_del)))
        self._save_role(role, to_add, to_del, ok=0, fail=0)

    def _save_role(self, role, to_add, to_del, ok, fail):
        if to_add:
            pid = to_add[0]
            try:
                r = self.api.session.post(self.api._url("/roles/%s/permissions" % role["id"]),
                                          json={"permission_id": pid}, timeout=15)
                r.raise_for_status()
                ok += 1
            except Exception:
                fail += 1
            self._save_role(role, to_add[1:], to_del, ok, fail)
            return
        if to_del:
            pid = to_del[0]
            try:
                r = self.api.session.delete(
                    self.api._url("/roles/%s/permissions/%s" % (role["id"], pid)), timeout=15)
                r.raise_for_status()
                ok += 1
            except Exception:
                fail += 1
            self._save_role(role, [], to_del[1:], ok, fail)
            return
        # всё применено
        self._edits[role["id"]] = set(role["perms"])  # перечитаем ниже
        msg = "Сохранено. Применено изменений: %d" % ok
        if fail:
            msg += ", ошибок: %d (проверьте право roles.manage)" % fail
        self.status.setText(msg)
        self.refresh()

"""FitIntel Pro — Generic Form Dialog"""
from PyQt6.QtWidgets import (
    QDialog, QVBoxLayout, QFormLayout, QLineEdit, QComboBox, QDialogButtonBox
)


class FormDialog(QDialog):
    """fields: list of (key, label, kind, options_or_default)
    kind: "text" | "combo"; combo options: list of (label, value)"""

    def __init__(self, title: str, fields: list, parent=None):
        super().__init__(parent)
        self.setWindowTitle(title)
        self.setMinimumWidth(420)
        self._widgets = {}
        layout = QVBoxLayout(self)
        form = QFormLayout()
        for f in fields:
            key, label = f[0], f[1]
            kind = f[2] if len(f) > 2 else "text"
            if kind == "combo":
                w = QComboBox()
                for opt in (f[3] if len(f) > 3 else []):
                    if isinstance(opt, tuple):
                        w.addItem(str(opt[0]), opt[1])
                    else:
                        w.addItem(str(opt), opt)
            else:
                w = QLineEdit()
                if len(f) > 3 and f[3] not in (None, ""):
                    w.setText(str(f[3]))
            w.setStyleSheet("padding: 6px; border: 1px solid #cbd5e1; border-radius: 4px;")
            self._widgets[key] = (kind, w)
            form.addRow(label, w)
        layout.addLayout(form)
        bb = QDialogButtonBox(QDialogButtonBox.StandardButton.Ok | QDialogButtonBox.StandardButton.Cancel)
        bb.accepted.connect(self.accept)
        bb.rejected.connect(self.reject)
        layout.addWidget(bb)

    def values(self) -> dict:
        out = {}
        for key, (kind, w) in self._widgets.items():
            out[key] = w.currentData() if kind == "combo" else w.text().strip()
        return out

"""FitIntel Pro — клиентское логирование (logs/client_YYYYMMDD.log)"""
import logging
import sys

from datetime import datetime
from pathlib import Path

LOG_DIR = Path(__file__).parent / "logs"
LOG_DIR.mkdir(exist_ok=True)
LOG_FILE = LOG_DIR / f"client_{datetime.now():%Y%m%d}.log"

logger = logging.getLogger("fitintel")
logger.setLevel(logging.DEBUG)
if not logger.handlers:
    fh = logging.FileHandler(LOG_FILE, encoding="utf-8")
    fh.setFormatter(logging.Formatter("%(asctime)s [%(levelname)s] %(message)s"))
    logger.addHandler(fh)

log = logger


def install_excepthook():
    def hook(exc_type, exc, tb):
        log.critical("UNHANDLED: %s", "".join(traceback.format_exception(exc_type, exc, tb)))
    sys.excepthook = hook
